/**
* Detect current network purely from hostname.
* Everything that is not explicitly mainnet defaults to testnet (pre-mainnet phase).
*/
export function detectNetwork(): NetworkId {
if (typeof window === "undefined") return "testnet";

const host = window.location.hostname.toLowerCase();

if (host === "mainnet.kovanica.online" || host.startsWith("mainnet.")) {
  return "mainnet";
}

return "testnet";
}

export function getCurrentNetwork(): NetworkConfig {
return NETWORKS[detectNetwork()];
}

/**
* Hard redirect to the target network, preserving path + query + hash.
*/
export function switchNetwork(target: NetworkId) {
const config = NETWORKS[target];
const path =
  window.location.pathname +
  window.location.search +
  window.location.hash;

window.location.href = `https://${config.hostname}${path}`;
}
